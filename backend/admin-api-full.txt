const express = require("express");
const multer = require("multer");

const { authenticate, authorize } = require("../../middlewares/auth.middleware");
const validate = require("../../middlewares/validate.middleware");

const categoryController = require("../../modules/catalog/category/category.controller");
const {
    categoryCreateSchema,
    categoryUpdateSchema,
} = require("../../modules/catalog/category/category.validator");

const brandController = require("../../modules/catalog/brand/brand.controller");
const {
    brandCreateSchema,
    brandUpdateSchema,
} = require("../../modules/catalog/brand/brand.validator");

const productController = require("../../modules/catalog/product/product.controller");
const {
    productCreateSchema,
    productUpdateSchema,
    productStatusUpdateSchema,
    productListQuerySchema,
    variantSchema,
    variantUpdateSchema,
} = require("../../modules/catalog/product/product.validator");

const { uploadCatalogImage } = require("../../modules/catalog/catalog-upload.controller");

const router = express.Router();

const imageUpload = multer({
    storage: multer.memoryStorage(),
    limits: {
        fileSize: 5 * 1024 * 1024,
        files: 1,
    },
    fileFilter(req, file, callback) {
        const allowedMimeTypes = [
            "image/jpeg",
            "image/png",
            "image/webp",
        ];

        if (!allowedMimeTypes.includes(file.mimetype)) {
            return callback(new Error("INVALID_IMAGE_TYPE"));
        }

        callback(null, true);
    },
});

const inventoryController = require("../../modules/inventory/inventory.controller");

const {
    inventoryListQuerySchema,
    inventoryVariantParamsSchema,
    inventoryMovementQuerySchema,
    inventoryPurchaseSchema,
    inventoryAdjustmentSchema,
} = require("../../modules/inventory/inventory.validator");

// All catalog management routes require an admin.
router.use(authenticate, authorize("ADMIN"));

// ==========================================
// CATALOG IMAGE UPLOAD
// ==========================================

router.post(
    "/images/:type",
    (req, res, next) => {
        imageUpload.single("image")(req, res, (error) => {
            if (!error) return next();

            if (
                error instanceof multer.MulterError &&
                error.code === "LIMIT_FILE_SIZE"
            ) {
                return res.status(400).json({
                    success: false,
                    message: "IMAGE_TOO_LARGE",
                });
            }

            return res.status(400).json({
                success: false,
                message: error.message === "INVALID_IMAGE_TYPE"
                    ? "INVALID_IMAGE_TYPE"
                    : "IMAGE_UPLOAD_INVALID",
            });
        });
    },
    uploadCatalogImage
);

// ==========================================
// CATEGORIES
// ==========================================

router.get("/categories", categoryController.listCategories);
router.get("/categories/:id", categoryController.getCategory);

router.post(
    "/categories",
    validate(categoryCreateSchema),
    categoryController.createCategory
);

router.patch(
    "/categories/:id",
    validate(categoryUpdateSchema),
    categoryController.updateCategory
);

router.delete("/categories/:id", categoryController.deleteCategory);

// ==========================================
// BRANDS

router.get("/brands", brandController.listBrands);

router.post(

    brandController.createBrand

router.patch(

// ==========================================
// PRODUCTS
// ==========================================

router.get(
    "/products",
    validate(productListQuerySchema, "query"),
    productController.listProducts

router.get(
    "/products/:id",
    productController.getProduct

router.post(

const express = require("express");

const {
    authenticate,
    authorize,
} = require("../../middlewares/auth.middleware");

const validate = require("../../middlewares/validate.middleware");

const controller = require("./member-profile-request.controller");
const validator = require("./member-profile-request.validator");

// ==========================================
// CUSTOMER ROUTER
// ==========================================

const customerRouter = express.Router();

customerRouter.use(
    authenticate,
    authorize("CUSTOMER")
);

// Create profile update request
customerRouter.post(
    "/",
    validate(validator.createProfileRequestSchema),
    controller.createMyProfileRequest
);

// List own profile update requests
customerRouter.get(
    "/",
    validate(validator.myProfileRequestListQuerySchema, "query"),
    controller.listMyProfileRequests
);

// Get own profile update request
customerRouter.get(
    "/:requestId",
    validate(validator.requestIdParamsSchema, "params"),
    controller.getMyProfileRequest
);

// ==========================================
// ADMIN ROUTER
// ==========================================

const adminRouter = express.Router();

adminRouter.use(
    authenticate,
    authorize("ADMIN")
);

// List all profile update requests
adminRouter.get(
    "/",
    validate(validator.adminProfileRequestListQuerySchema, "query"),
    controller.listAdminProfileRequests
);

// Get profile update request detail
adminRouter.get(
    "/:requestId",
    validate(validator.requestIdParamsSchema, "params"),
    controller.getAdminProfileRequest
);

// Approve profile update request
adminRouter.post(
    "/:requestId/approve",
    validate(validator.requestIdParamsSchema, "params"),
    validate(validator.approveProfileRequestSchema),
    controller.approveProfileRequest
);

// Reject profile update request
adminRouter.post(
    "/:requestId/reject",
    validate(validator.requestIdParamsSchema, "params"),
    validate(validator.rejectProfileRequestSchema),
    controller.rejectProfileRequest
);

// ==========================================
// EXPORTS
// ==========================================

module.exports = {
    customerProfileRequestRouter: customerRouter,
    adminProfileRequestRouter: adminRouter,
};
